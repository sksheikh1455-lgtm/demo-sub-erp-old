          .sort((a: any, b: any) => String(b.date || '').localeCompare(String(a.date || '')));

          const item = (b.items || []).find((it: any) => it.productId === productId);

        return null;

        productId: productId, 
        description: p?.name || '', 
        unitPrice: lastRate !== null ? lastRate : (p?.price || 0), 
        serialNumbers: [] 
      };
      return {...prev, items: ni};
    });
